# ====================================================================
# TVLA analysis (post-capture)
# ====================================================================
welch_t analysis
    scripts:
        pre-silicon-mlkem-tvla/scripts/silicon/validate_silicon_ccapc.py
        pre-silicon-mlkem-tvla/scripts/L1/validate_L1.py
        pre-silicon-mlkem-tvla/scripts/L35/validate_L35.py
        pre-silicon-mlkem-tvla/scripts/L35/enumerate_leaky_pips_full.py
    pythondeps:
        numpy
        scipy.stats
    methodology: Welch's t-test, sample-by-sample, post-trigger max|t[250:]|
                 (silicon) or per (FF, cycle, bit) (L1) or per (cycle, tile) (L3.5)
    thresholds:  4.5σ (TVLA convention) or Bonferroni-corrected α=1e-5
                 (~5.85σ for L1, ~6.51σ when n_comparisons=134,400)
